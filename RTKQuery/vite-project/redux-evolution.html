1. Vanilla Redux

~2015 — The original
Manual everything No async built-in ~80 lines
Action Types
// constants.js
const FETCH_REQUEST  = 'FETCH_REQUEST'
const FETCH_SUCCESS  = 'FETCH_SUCCESS'
const FETCH_FAILURE  = 'FETCH_FAILURE'
const ADD_REQUEST    = 'ADD_REQUEST'
const ADD_SUCCESS    = 'ADD_SUCCESS'
const ADD_FAILURE    = 'ADD_FAILURE'
Reducer
// reducer.js
const init = { posts:[], loading:false, error:null }

function reducer(state = init, action) {
  switch(action.type) {
    case FETCH_REQUEST:
      return { ...state, loading:true }
    case FETCH_SUCCESS:
      return { ...state, loading:false,
               posts: action.payload }
    case FETCH_FAILURE:
      return { ...state, loading:false,
               error: action.payload }
    default: return state
  }
}
Store
import { createStore } from 'redux'
const store = createStore(reducer)
Component
// async lives IN the component
function Posts() {
  const dispatch = useDispatch()
  const { posts, loading } = useSelector(s => s)

  const load = async () => {
    dispatch({ type: FETCH_REQUEST })
    try {
      const res  = await fetch('/posts')
      const data = await res.json()
      dispatch({ type: FETCH_SUCCESS, payload: data })
    } catch(e) {
      dispatch({ type: FETCH_FAILURE, payload: e.message })
    }
  }

  useEffect(() => { load() }, [])

  if (loading) return <p>Loading...</p>
  return <ul>{posts.map(p =>
    <li key={p.id}>{p.title}</li>
  )}</ul>
}
😟 6 action type strings, manual try/catch in component, spread every state update, easy to forget loading/error
~80 lines of boilerplate for one feature

2. Redux + Thunk

~2016 — Async moved to action creators
Thunk middleware Async in actions ~70 lines
Action Types (same 6 constants)
// Still need all 6 string constants...
const FETCH_REQUEST = 'FETCH_REQUEST'
// ... etc
Thunk Action Creators
// actions.js — async lives here now
const fetchPosts = () => async (dispatch) => {
  dispatch({ type: FETCH_REQUEST })
  try {
    const res  = await fetch('/posts')
    const data = await res.json()
    dispatch({ type: FETCH_SUCCESS, payload: data })
  } catch(e) {
    dispatch({ type: FETCH_FAILURE, payload: e.message })
  }
}

const addPost = (post) => async (dispatch) => {
  dispatch({ type: ADD_REQUEST })
  try {
    const res  = await fetch('/posts', {
      method: 'POST', body: JSON.stringify(post)
    })
    const data = await res.json()
    dispatch({ type: ADD_SUCCESS, payload: data })
    dispatch(fetchPosts()) // manual refetch!
  } catch(e) {
    dispatch({ type: ADD_FAILURE, payload: e.message })
  }
}
Store
import { createStore, applyMiddleware } from 'redux'
import thunk from 'redux-thunk'

const store = createStore(
  reducer,
  applyMiddleware(thunk)
)
Component (cleaner)
// Posts.jsx — no async here now!
function Posts() {
  const dispatch = useDispatch()
  const { posts, loading } = useSelector(s => s)

  useEffect(() => {
    dispatch(fetchPosts()) // just dispatch
  }, [])

  if (loading) return <p>Loading...</p>
  return <ul>{posts.map(p =>
    <li key={p.id}>{p.title}</li>
  )}</ul>
}
✅ Async out of component
😟 Still: 6 string constants, switch/case, spread operators, manual refetch after add
~70 lines — marginal improvement

3. RTK (Redux Toolkit)

~2019 — Official opinionated Redux
createSlice createAsyncThunk Immer built-in ~40 lines
Slice (replaces constants + reducer)
// postsSlice.js
import { createSlice, createAsyncThunk }
  from '@reduxjs/toolkit'

// auto-generates pending/fulfilled/rejected
export const fetchPosts = createAsyncThunk(
  'posts/fetch',
  async () => {
    const res = await fetch('/posts')
    return res.json() // no try/catch needed!
  }
)

export const addPost = createAsyncThunk(
  'posts/add',
  async (post, { dispatch }) => {
    const res = await fetch('/posts', {
      method: 'POST', body: JSON.stringify(post)
    })
    const data = await res.json()
    dispatch(fetchPosts()) // still manual refetch
    return data
  }
)

const slice = createSlice({
  name: 'posts',
  initialState: { posts:[], loading:false, error:null },
  reducers: {},
  extraReducers: (builder) => {
    builder
      .addCase(fetchPosts.pending,   (state) => {
        state.loading = true  // Immer: mutate directly!
      })
      .addCase(fetchPosts.fulfilled, (state, { payload }) => {
        state.loading = false
        state.posts   = payload
      })
      .addCase(fetchPosts.rejected,  (state, { error }) => {
        state.loading = false
        state.error   = error.message
      })
  }
})
export default slice.reducer
Store
import { configureStore } from '@reduxjs/toolkit'
// thunk middleware included by default!
export const store = configureStore({
  reducer: { posts: postsReducer }
})
Component
function Posts() {
  const dispatch = useDispatch()
  const { posts, loading } =
    useSelector(s => s.posts)

  useEffect(() => {
    dispatch(fetchPosts())
  }, [])

  const handleAdd = () =>
    dispatch(addPost({ title:'New', userId:1 }))

  if (loading) return <p>Loading...</p>
  return (
    <>
      <button onClick={handleAdd}>Add</button>
      <ul>{posts.map(p =>
        <li key={p.id}>{p.title}</li>
      )}</ul>
    </>
  )
}
✅ No string constants, no spread, Immer direct mutation, auto pending/fulfilled/rejected
😟 Still: manage loading/error yourself, useEffect to trigger, no caching, manual refetch
~40 lines — big improvement!

4. RTK Query

~2021 — Data fetching fully solved
Auto caching Auto loading states Auto refetch ~15 lines
API Slice (everything in one place)
// apiSlice.js
import { createApi, fetchBaseQuery }
  from '@reduxjs/toolkit/query/react'

export const apiSlice = createApi({
  reducerPath: 'api',
  baseQuery: fetchBaseQuery({
    baseUrl: 'https://jsonplaceholder.typicode.com'
  }),
  tagTypes: ['Post'],
  endpoints: (builder) => ({

    getPosts: builder.query({
      query: () => '/posts?_limit=5',
      providesTags: ['Post']
    }),

    addPost: builder.mutation({
      query: (post) => ({
        url: '/posts', method: 'POST', body: post
      }),
      invalidatesTags: ['Post'] // auto refetch!
    }),

  })
})

export const {
  useGetPostsQuery,
  useAddPostMutation
} = apiSlice
Store
import { configureStore } from '@reduxjs/toolkit'
import { apiSlice } from './apiSlice'

export const store = configureStore({
  reducer: {
    [apiSlice.reducerPath]: apiSlice.reducer
  },
  middleware: (get) =>
    get().concat(apiSlice.middleware)
})
Component (minimal!)
// No useEffect. No dispatch. No loading state.
function Posts() {
  const {
    data: posts = [],
    isLoading, isError
  } = useGetPostsQuery()
  // fetches on mount, caches, deduplicates

  const [addPost, { isLoading: isAdding }]
    = useAddPostMutation()

  if (isLoading) return <p>Loading...</p>
  if (isError)   return <p>Error!</p>

  return (
    <>
      <button
        disabled={isAdding}
        onClick={() => addPost({ title:'New', userId:1 })}
      >
        {isAdding ? 'Adding...' : 'Add Post'}
      </button>
      <ul>{posts.map(p =>
        <li key={p.id}>{p.title}</li>
      )}</ul>
    </>
  )
}
✅ Zero loading/error management, auto caching, auto refetch on mutation, no useEffect, no dispatch for fetching, request deduplication
~15 lines in component — 5x less than vanilla!
